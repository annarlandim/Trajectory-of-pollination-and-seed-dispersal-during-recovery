model{
    # theta_inf, tau_old over J
    mean_theta_inf ~ dnorm(0, 1)
    sigma_theta_inf ~ dscaled.gamma(1, 2) # re-scale it with sigma
    for(j in 1:n_var) {
      beta_inf_raw[j] ~ dnorm(0, 1) 
      theta_inf[j] <- exp(mean_theta_inf + beta_inf_raw[j] * sigma_theta_inf)
      # priors on variance components
      tau_old[j] ~ dscaled.gamma(s_old[j], 2)
      sigmaSq_old[j] <- pow(tau_old[j], -1)
    }
    # loop for theta_0, tau_rec over J
    mean_theta_0 ~ dnorm(0, 1)
    sigma_theta_0 ~ dscaled.gamma(1, 2) # re-scale it with sigma
    for(j in 1:n_var) {
      beta_0_raw[j] ~ dnorm(0, 1) 
      theta_0[j] <- exp(mean_theta_0 + beta_0_raw[j] * sigma_theta_0)
      tau_rec[j] ~ dscaled.gamma(s_rec[j], 2)
      sigmaSq_rec[j] <- pow(tau_rec[j], -1) # added by Anna
    }
    
    # loop for alpha_con & beta_con over J
    mean_alpha_con ~ dnorm(0, 1)
    sigma_alpha_con ~ dscaled.gamma(1, 2) # re-scale it with sigma
    mean_beta_con ~ dnorm(0, 1)
    sigma_beta_con ~ dscaled.gamma(1, 2) # re-scale it with sigma
    for(j in 1:n_var) {
      alpha_con_raw[j] ~ dnorm(0, 1)
      alpha_con[j] <- mean_alpha_con + alpha_con_raw[j] * sigma_alpha_con
      beta_con_raw[j] ~ dnorm(0, 1)
      beta_con[j] <- mean_beta_con + beta_con_raw[j] * sigma_beta_con
    }
    # Likelihood part of the model
    for (i in 1:n_old) {
      # Model Likelihood for old-growth forests
      Y_old[i] ~ dlnorm(log(theta_inf[variable_old[i]]), tau_old[variable_old[i]]) # check with jorg if not having log(theta_inf) was a mistake or on purpose
    }
    for (i in 1:n_rec) {							
      mu_rec[i] <-
        theta_0[variable_rec[i]] + 
        (theta_inf[variable_rec[i]] - theta_0[variable_rec[i]]) *
        (1 - exp(-lambda[i] * tx[i]))
      # Model lambda depending on connectivity:
      lambda[i] <- exp(alpha_con[variable_rec[i]] + beta_con[variable_rec[i]] * connectivity[i])
      # Model Likelihood for secondary forests
      Y_rec[i] ~ dlnorm(log(mu_rec[i]), tau_rec[variable_rec[i]])
    }
  }
